   617→function AddCourseForm({ onCreated }: { onCreated: () => void }) {
   618→  const [form, setForm] = useState({ code: "", title: "", description: "", duration: "", deliveryMethod: "", entryRequirements: "" });
   619→  const [error, setError] = useState<string | null>(null);
   620→  const [creating, setCreating] = useState(false);
   621→  const handleCreate = async (e: React.FormEvent) => {
   622→    e.preventDefault(); setCreating(true); setError(null);
   623→    try {
   624→      const res = await fetch("/api/academy/courses", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "create", ...form }) });
   625→      const data = await res.json(); if (!res.ok || !data.ok) throw new Error(data.error);
   626→      onCreated();
   627→    } catch (err) { setError(err instanceof Error ? err.message : "Failed."); } finally { setCreating(false); }
   628→  };
