   409→   409→function ServiceCard({
   410→   410→  service,
   411→   411→  index,
   412→   412→  onOpen,
   413→   413→  animate,
   414→   414→}: {
   415→   415→  service: Service;
   416→   416→  index: number;
   417→   417→  onOpen: () => void;
   418→   418→  animate: boolean;
   419→   419→}) {
   420→   420→  const cardRef = useRef<HTMLDivElement>(null);
   421→   421→
   422→   422→  useEffect(() => {
   423→   423→    const el = cardRef.current;
   424→   424→    if (!el || !animate) return;
   425→   425→
   426→   426→    gsap.fromTo(
   427→   427→      el,
   428→   428→      { opacity: 0, y: 30 },
   429→   429→      {
   430→   430→        opacity: 1,
   431→   431→        y: 0,
   432→   432→        duration: 0.5,
   433→   433→        delay: index * 0.07,
   434→   434→        ease: "power3.out",
   435→   435→      }
   436→   436→    );
   437→   437→  }, [index, animate]);
   438→   438→
   439→   439→  const handleKeyDown = (e: React.KeyboardEvent) => {
   440→   440→    if (e.key === "Enter" || e.key === " ") {
   441→   441→      e.preventDefault();
   442→   442→      onOpen();
   443→   443→    }
   444→   444→  };
   445→   445→
   446→   446→  return (
   447→   447→    <div ref={cardRef} style={animate ? { opacity: 0 } : undefined}>
   448→   448→      <Card
   449→   449→        role="button"
   450→   450→        tabIndex={0}
   451→   451→        aria-label={`View details for ${service.title}`}
   452→   452→        onClick={onOpen}
   453→   453→        onKeyDown={handleKeyDown}
   454→   454→        className={`h-full bg-dark-card/80 backdrop-blur-sm border-dark-border/50 ${service.accentBorder} transition-all duration-500 overflow-hidden group relative cursor-pointer focus-visible:ring-2 focus-visible:ring-brand/50 focus-visible:ring-offset-2 focus-visible:ring-offset-dark-deep hover:scale-[1.02] hover:-translate-y-1 active:scale-[0.98]`}
   455→   455→      >
   456→   456→        {/* Top Accent Bar */}
   457→   457→        <div
   458→   458→          aria-hidden="true"
   459→   459→          className={`h-[2px] bg-gradient-to-r ${service.accent} opacity-30 group-hover:opacity-80 transition-opacity duration-500`}
   460→   460→        />
   461→   461→
   462→   462→        {/* Visible image header band */}
   463→   463→        <div className="card-image-header">
   464→   464→          <img src={service.image} alt="" />
   465→   465→          {/* Tag chip floating on the image */}
   466→   466→          <span
   467→   467→            className={`absolute top-3 left-3 z-10 text-[10px] tracking-[0.15em] uppercase font-bold ${service.tagColor} px-3 py-1.5 rounded-full backdrop-blur-md`}
   468→   468→          >
   469→   469→            {service.tag}
   470→   470→          </span>
   471→   471→        </div>
